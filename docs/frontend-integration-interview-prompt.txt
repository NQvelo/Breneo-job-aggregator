/** Play MP3 URLs in order; resolves when each segment ends. */
export function playAudioUrl(url: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const audio = new Audio(url);
    audio.onended = () => resolve();
    audio.onerror = () => reject(new Error(`Audio failed: ${url}`));
    const playPromise = audio.play();
    if (playPromise) {
      playPromise.catch(reject); // autoplay blocked → show manual play button
    }
  });
}

export async function playPlaybackSequence(
  items: InterviewPlaybackItem[],
): Promise<void> {
  for (const item of items) {
    if (!item.audio_url) continue;
    await playAudioUrl(item.audio_url);
  }
}
